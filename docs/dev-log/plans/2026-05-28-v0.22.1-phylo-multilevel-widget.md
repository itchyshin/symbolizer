  library(dplyr)

# Fetch the canonical CSV + tree
src_url   <- "https://raw.githubusercontent.com/itchyshin/location-scale_meta-analysis/main/data"
csv_dest  <- tempfile(fileext = ".csv")
tree_dest <- tempfile(fileext = ".tre")
download.file(file.path(src_url, "thermal.csv"),    csv_dest)
download.file(file.path(src_url, "phylo_tree.tre"), tree_dest)

dat_full  <- read.csv(csv_dest, stringsAsFactors = FALSE)

# Pick 35 species at random; keep ALL their effects; cap at 250 effects
species_pool <- intersect(unique(dat_full$species_ID), tree_full$tip.label)
species_keep <- sample(species_pool, 35L)
dat_sub <- dat_full %>%
  filter(species_ID %in% species_keep) %>%
  group_by(species_ID) %>%
  slice_sample(n = pmin(n(), 10L)) %>%   # cap to 10 effects per species
  ungroup() %>%
  arrange(study_ID, species_ID)
if (nrow(dat_sub) > 250L) dat_sub <- dat_sub[seq_len(250L), ]

tree_sub <- ape::keep.tip(tree_full, species_keep)
tree_sub <- ape::chronos(tree_sub, lambda = 1)

dir.create("inst/extdata", showWarnings = FALSE, recursive = TRUE)
write.csv(dat_sub, "inst/extdata/thermal_subset.csv", row.names = FALSE)
ape::write.tree(tree_sub, "inst/extdata/thermal_subset_tree.tre")

cat(sprintf("Wrote %d effects across %d species across %d studies.\n",
            nrow(dat_sub),
            length(unique(dat_sub$species_ID)),
            length(unique(dat_sub$study_ID))))
```

# Helper fixtures for v0.22.1 phylogenetic multilevel meta-analysis
# tests. All three fixtures use the deterministic 250-effect subsample
# of thermal.csv shipped at inst/extdata/thermal_subset.csv.

.read_thermal_subset <- function() {
  csv_path <- system.file("extdata", "thermal_subset.csv",
                          package = "symbolizer")
  if (!nzchar(csv_path)) {
    csv_path <- "inst/extdata/thermal_subset.csv"   # dev-mode fallback
  }
  read.csv(csv_path, stringsAsFactors = FALSE)
}

.read_thermal_tree <- function() {
  testthat::skip_if_not_installed("ape")
  tree_path <- system.file("extdata", "thermal_subset_tree.tre",
                           package = "symbolizer")
  if (!nzchar(tree_path)) {
    tree_path <- "inst/extdata/thermal_subset_tree.tre"
  }
  ape::read.tree(tree_path)
}

fit_drmtmb_phylo_multilevel <- function() {
  testthat::skip_if_not_installed("drmTMB")
  dat  <- .read_thermal_subset()
  tree <- .read_thermal_tree()
  drmTMB::drmTMB(
    drmTMB::drm_formula(
      dARR ~ 1 + habitat +
              drmTMB::phylo(1 | species_ID, tree = tree) +
              (1 | study_ID),
      family = stats::gaussian()),
    data    = dat,
    meta_V  = drmTMB::meta_V(V = dat$Var_dARR),
    silent  = TRUE
  )
}

fit_brms_phylo_meta <- function() {
  testthat::skip_if_not_installed("brms")
  testthat::skip_if_not_installed("ape")
  dat  <- .read_thermal_subset()
  tree <- .read_thermal_tree()
  A_phylo <- ape::vcv.phylo(tree, corr = TRUE)
  brms::brm(
    brms::bf(dARR | brms::se(sqrt(Var_dARR)) ~ 1 + habitat
                + (1 | species_ID) + (1 | study_ID)
                + (1 | gr(species_ID, cov = mat))),
    data   = dat,
    data2  = list(mat = A_phylo),
    family = gaussian(),
    chains = 2L, iter = 1000L, warmup = 500L,
    refresh = 0L, silent = 2,
    file = "/tmp/fit_brms_phylo_meta_cache"
  )
}

fit_metafor_phylo_meta <- function() {
  testthat::skip_if_not_installed("metafor")
  testthat::skip_if_not_installed("ape")
  dat  <- .read_thermal_subset()
  tree <- .read_thermal_tree()
  A_phylo <- ape::vcv.phylo(tree, corr = TRUE)
  dat$phylo <- dat$species_ID
  metafor::rma.mv(
    yi     = dARR,
    V      = Var_dARR,
    mods   = ~ 1 + habitat,
    random = list(~ 1 | species_ID, ~ 1 | study_ID, ~ 1 | phylo),
    R      = list(phylo = A_phylo),
    data   = dat
  )
}
```

# is the presence of `se(sqrt(...))` or `se(...)` in the response

brms_response_text <- tryCatch({
  rf <- fit$formula$forms[[1L]]$response
  if (!is.null(rf)) paste(deparse(rf), collapse = " ") else ""

if (!nzchar(brms_response_text)) {
  brms_response_text <- paste(deparse(fit$formula), collapse = " ")
}
has_se <- grepl("\\bse\\s*\\(", brms_response_text)
metadata$context <- if (isTRUE(has_se)) "meta_analysis" else NA_character_

```markdown
## 4. Phylogenetic multilevel meta-analysis

When studies are on related species, the between-study tier splits
into a phylogenetic piece (constrained by the phylogeny via
$\mathbf{A}$) and three non-phylogenetic tiers (effect-level,
study-level, species-level).

### 4.1 The model in symbols

For each effect $kt$ in study $k$ on species $s(k)$:

$$
y_{kt} = \beta_0 + \beta_1 x_{kt} + u_{p,s(k)} + u_{s,s(k)} + u_{\text{study},k} + u_{\text{es},kt} + \epsilon_{kt}
$$

with

$$
u_{p,s} \sim \mathcal{N}(\mathbf{0}, \sigma_p^2 \mathbf{A}),
\;
u_{s,s} \sim \mathcal{N}(0, \sigma_s^2),
\;
u_{\text{study},k} \sim \mathcal{N}(0, \sigma_{\text{study}}^2),
\;
u_{\text{es},kt} \sim \mathcal{N}(0, \sigma_{\text{es}}^2),
\;
\epsilon_{kt} \sim \mathcal{N}(0, v_{kt}).
$$

The marginal variance of one effect is

$$
\mathrm{Var}(y_{kt}) = v_{kt} + \sigma_{\text{es}}^2 + \sigma_{\text{study}}^2 + \sigma_s^2 + \sigma_p^2 \mathbf{A}_{s(k), s(k)}.
$$
